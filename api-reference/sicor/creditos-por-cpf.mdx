---
title: "Financiamentos Rurais por CPF/CNPJ"
api: "GET /sicor/creditos/cpf/{cpf_cnpj}"
description: "Consulta financiamentos e operacoes de credito rural vinculadas a um CPF ou CNPJ"
---

## Descricao

Retorna todas as operacoes de credito rural registradas no BCB vinculadas ao CPF ou CNPJ informado. Util para verificar o historico de credito rural de um produtor ou empresa.

<ParamField path="cpf_cnpj" type="string" required>
  CPF (11 digitos) ou CNPJ (14 digitos), apenas numeros. Aceita com ou sem formatacao.
</ParamField>

<ParamField query="offset" type="integer" default={0}>
  Offset para paginacao.
</ParamField>

<ParamField query="limit" type="integer" default={100}>
  Quantidade por pagina (max: 500).
</ParamField>

## Response

<ResponseField name="data" type="object">
  <Expandable title="data">
    <ResponseField name="cpf_cnpj" type="string">
      CPF/CNPJ consultado.
    </ResponseField>

    <ResponseField name="resumo" type="object">
      Resumo agregado.
      <Expandable title="resumo">
        <ResponseField name="total_operacoes" type="integer">
          Total de operacoes.
        </ResponseField>
        <ResponseField name="primeira_operacao" type="string">
          Data da primeira operacao.
        </ResponseField>
        <ResponseField name="ultima_operacao" type="string">
          Data da ultima operacao.
        </ResponseField>
        <ResponseField name="valor_total_credito" type="number">
          Soma dos valores de credito (R$).
        </ResponseField>
        <ResponseField name="area_total_financiada" type="number">
          Soma das areas financiadas (ha).
        </ResponseField>
        <ResponseField name="total_cars" type="integer">
          CARs distintos vinculados a esse CPF/CNPJ.
        </ResponseField>
        <ResponseField name="operacoes_vigentes" type="integer">
          Operacoes com vencimento futuro.
        </ResponseField>
        <ResponseField name="operacoes_vencidas" type="integer">
          Operacoes ja vencidas.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="operacoes" type="array">
      Lista de operacoes (mesmo formato de `/sicor/creditos/car`).
    </ResponseField>

    <ResponseField name="metadata" type="object">
      Metadados de paginacao.
    </ResponseField>
  </Expandable>
</ResponseField>
